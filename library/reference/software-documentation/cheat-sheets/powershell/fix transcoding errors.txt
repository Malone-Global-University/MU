$path = "C:\MGU\library\reference\software-documentation\cheat-sheets\powershell\index.html"

$utf8 = [System.Text.UTF8Encoding]::new($false, $true)
$content = [System.IO.File]::ReadAllText($path, $utf8)

$bad = -join ([char]0x00E2, [char]0x20AC, [char]0x201D)
$good = [string][char]0x2014

$count = [regex]::Matches($content, [regex]::Escape($bad)).Count

if ($count -gt 0) {
    $content = $content.Replace($bad, $good)
    [System.IO.File]::WriteAllText($path, $content, $utf8)
    Write-Host "SUCCESS: Corrected $count corrupted characters." -ForegroundColor Green
} else {
    Write-Host "No matching corrupted characters found." -ForegroundColor Yellow
}
